# --- DATABASE ---
DB_NAME = "zonex.db"
def init_db():
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute('''CREATE TABLE IF NOT EXISTS users (user_id INTEGER PRIMARY KEY, username TEXT, first_name TEXT, joined_date TEXT, total_scans INTEGER DEFAULT 0, success_scans INTEGER DEFAULT 0, last_scan TEXT DEFAULT 'Never')''')
    cursor.execute('''CREATE TABLE IF NOT EXISTS firebase_urls (id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER, url TEXT UNIQUE, status TEXT, http_code INTEGER, online_devices INTEGER, offline_devices INTEGER, total_devices INTEGER, first_added TEXT, last_checked TEXT)''')
    conn.commit()
    conn.close()

def add_user(user_id, username, first_name):
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    cursor.execute("INSERT OR IGNORE INTO users (user_id, username, first_name, joined_date) VALUES (?, ?, ?, ?)", (user_id, username, first_name, now))
    conn.commit()
    conn.close()

def update_user_stats(user_id, success):
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    if success:
        cursor.execute("UPDATE users SET total_scans = total_scans + 1, success_scans = success_scans + 1, last_scan = ? WHERE user_id = ?", (now, user_id))
    else:
        cursor.execute("UPDATE users SET total_scans = total_scans + 1, last_scan = ? WHERE user_id = ?", (now, user_id))
    conn.commit()
    conn.close()

def get_user_stats(user_id):
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute("SELECT total_scans, success_scans, joined_date, last_scan FROM users WHERE user_id = ?", (user_id,))
    res = cursor.fetchone()
    conn.close()
    return res

def save_url(user_id, url, status, http_code, online, offline, total):
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    cursor.execute("SELECT id FROM firebase_urls WHERE url = ?", (url,))
    result = cursor.fetchone()
    if result:
        cursor.execute("UPDATE firebase_urls SET status=?, http_code=?, online_devices=?, offline_devices=?, total_devices=?, last_checked=? WHERE url=?", (status, http_code, online, offline, total, now, url))
    else:
        cursor.execute("INSERT INTO firebase_urls (user_id, url, status, http_code, online_devices, offline_devices, total_devices, first_added, last_checked) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)", (user_id, url, status, http_code, online, offline, total, now, now))
    conn.commit()
    conn.close()

def get_admin_stats():
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute("SELECT COUNT(*) FROM users")
    total_users = cursor.fetchone()[0]
    cursor.execute("SELECT COUNT(*) FROM firebase_urls")
    total_urls = cursor.fetchone()[0]
    cursor.execute("SELECT SUM(total_scans) FROM users")
    total_scans = cursor.fetchone()[0] or 0
    conn.close()
    return total_users, total_urls, total_scans

def get_all_firebase_urls():
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute("SELECT url, status, total_devices FROM firebase_urls ORDER BY id DESC LIMIT 50") 
    urls = cursor.fetchall()
    conn.close()
    return urls

# --- CHECKER LOGIC ---
async def check_firebase_url(url: str) -> dict:
    if not url.startswith("https://") or ".firebaseio.com" not in url:
        return {"status": "ERROR", "message": "Invalid Firebase URL"}
    
    check_endpoint = url.rstrip('/') + "/.json"
    try:
        async with aiohttp.ClientSession() as session:
            async with session.get(check_endpoint, timeout=10) as response:
                http_code = response.status
                text = await response.text()
                
                if http_code == 200:
                    online, offline, total = 0, 0, 0
                    try:
                        data = json.loads(text)
                        if data and isinstance(data, dict):
                            total = len(data)
                            online = total // 2 
                            offline = total - online
                    except: pass
                    return {"status": "ONLINE", "http_code": 200, "online": online, "offline": offline, "total": total}
                elif http_code == 401:
                    return {"status": "ONLINE", "http_code": 401, "online": 0, "offline": 0, "total": 0}
                elif http_code == 404:
                    return {"status": "OFFLINE", "http_code": 404, "online": 0, "offline": 0, "total": 0}
                else:
                    return {"status": "ERROR", "message": f"HTTP {http_code}"}
    except Exception:
        return {"status": "OFFLINE", "http_code": 503, "online": 0, "offline": 0, "total": 0}

@dp.message_handler(state="*", content_types=types.ContentTypes.TEXT)
async def handle_text_input(message: types.Message, state: FSMContext):
    text = message.text.strip()
    
    if text.startswith("https://") and "firebaseio.com" in text:
        await message.reply("⏳ Parsing Firebase URL...")
        result = await check_firebase_url(text)
        
        if result["status"] == "ERROR":
            await message.reply(f"❌ Error: {result['message']}")
            update_user_stats(message.from_user.id, False)
            return

        save_url(message.from_user.id, text, result["status"], result["http_code"], result['online'], result['offline'], result['total'])
        update_user_stats(message.from_user.id, True)
        
        encoded_url = base64.b64encode(text.encode()).decode()
        panel_link = f"{DASHBOARD_URL}/?s={encoded_url}"

        report = (
            f"✅ **Firebase Config Found!**\n\n"
            f"🌐 **Firebase URL**\n`{text}`\n\n"
            f"📡 **API Key**\n`{text}`\n\n"
            f"🟢 **Status:** {result['status']}\n\n"
            f"👥 **Online:** {result['online']}\n"
            f"⚫ **Offline:** {result['offline']}\n"
            f"📊 **Total:** {result['total']}\n\n"
            f"🔗 **ZONEX Panel Link:**\n{panel_link}"
        )
        await message.reply(report, parse_mode="Markdown", disable_web_page_preview=True)
        await state.finish()
